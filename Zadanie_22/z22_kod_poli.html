    <input type="button" name="pole_obwod" value="Oblicz pole oraz obwód" onclick="pole_obwod()"><br><br><br>

    <p id="pole"></p>
    <p id="obwod"></p>
    <script type="text/javascript">
        function pole_obwod() {
            pole_polishchuk();
            obwod_polishchuk();
        }
        function pole_polishchuk() {
            let a_dana_pol = parseFloat(document.querySelector("#a_dana").value);
            let b_dana_pol = parseFloat(document.querySelector("#b_dana").value);
            const p = document.querySelector("#pole");
            p.textContent = "Pole prostokąta = " + a_dana_pol*b_dana_pol + " cm^2";
        }
        function obwod_polishchuk() {
            let a_dana_pol = parseFloat(document.querySelector("#a_dana").value);
            let b_dana_pol = parseFloat(document.querySelector("#b_dana").value);
            const p = document.querySelector("#obwod");
            p.textContent = "Obwod prostokąta = " + (a_dana_pol*2 + b_dana_pol*2) + " cm";
        }
    </script>
