    <input type="button" name="boki" value="Oblicz wszystkie boki" onclick="boki()"><br><br>


    <p id="komunikat_AB"></p>
    <p id="komunikat_BC"></p>
    <p id="komunikat_AC"></p>
    <script type="text/javascript">
        function boki() {
            AB_pol();
            BC_pol();
            AC_pol();
        }
        function AB_pol() {
            let Ax_pol = parseFloat(document.getElementById('Ax').value);
            let Ay_pol = parseFloat(document.getElementById('Ay').value);
            let Bx_pol = parseFloat(document.getElementById('Bx').value);
            let By_pol = parseFloat(document.getElementById('By').value);
            const p = document.getElementById("komunikat_AB");
            p.textContent = "Długość boku AB = " + Math.sqrt(Math.pow((Ax_pol - Bx_pol), 2) + Math.pow((Ay_pol - By_pol), 2))
        }
        function BC_pol() {
            let Bx_pol = parseFloat(document.getElementById('Bx').value);
            let By_pol = parseFloat(document.getElementById('By').value);
            let Cx_pol = parseFloat(document.getElementById('Cx').value);
            let Cy_pol = parseFloat(document.getElementById('Cy').value);
            const p = document.getElementById("komunikat_BC");
            p.textContent = "Długość boku BC = " + Math.sqrt(Math.pow((Bx_pol - Cx_pol), 2) + Math.pow((By_pol - Cy_pol), 2))
        }
        function AC_pol() {
            let Ax_pol = parseFloat(document.getElementById('Ax').value);
            let Ay_pol = parseFloat(document.getElementById('Ay').value);
            let Cx_pol = parseFloat(document.getElementById('Cx').value);
            let Cy_pol = parseFloat(document.getElementById('Cy').value);
            const p = document.getElementById("komunikat_AC");
            p.textContent = "Długość boku AC = " + Math.sqrt(Math.pow((Ax_pol - Cx_pol), 2) + Math.pow((Ay_pol - Cy_pol), 2))
        }
    </script>
