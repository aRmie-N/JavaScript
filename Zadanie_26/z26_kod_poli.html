function poleTrapezu(a, b, h) {
    return ((a + b) * h) / 2;
}

function obwodTrapezuRownoramiennego(a, b, h) {
    const przesuniecie = Math.abs(a - b) / 2;
    const ramie = Math.sqrt(przesuniecie ** 2 + h ** 2);
    return a + b + 2 * ramie;
}

// Pole: (a + b) * h / 2
// Obwód: a + b + 2 * sqrt(h^2 + ((a - b) / 2)^2)
