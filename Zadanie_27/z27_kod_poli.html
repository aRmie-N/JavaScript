function obliczWysokosc(droga, odlegloscPozioma) {
    return Math.sqrt(droga ** 2 - odlegloscPozioma ** 2);
}

function obliczKatWznoszenia(droga, odlegloscPozioma) {
    const katWRadianach = Math.acos(odlegloscPozioma / droga);
    return katWRadianach * 180 / Math.PI;
}

// Wysokość wynika z twierdzenia Pitagorasa.
